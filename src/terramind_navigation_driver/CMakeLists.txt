cmake_minimum_required(VERSION 3.16)
project(terramind_navigation_driver)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(terramind_transport REQUIRED)
find_package(terramind_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(Threads REQUIRED)
add_library(navigation_core SHARED src/protocol.cpp src/coordinates.cpp src/session.cpp)
target_include_directories(navigation_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(navigation_core PUBLIC Eigen3::Eigen)
add_library(navigation_discovery SHARED src/port_discovery.cpp)
target_link_libraries(navigation_discovery PUBLIC navigation_core PRIVATE Threads::Threads)
ament_target_dependencies(navigation_discovery PUBLIC terramind_transport)
add_executable(navigation_serial_node src/navigation_serial_node.cpp)
target_link_libraries(navigation_serial_node navigation_discovery Threads::Threads)
ament_target_dependencies(navigation_serial_node rclcpp terramind_transport terramind_interfaces geometry_msgs nav_msgs diagnostic_msgs)
install(TARGETS navigation_core navigation_discovery EXPORT export_navigation_core LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS navigation_serial_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS tools/navigation_integration_test.py tools/navigation_discovery_integration_test.py DESTINATION lib/${PROJECT_NAME})
ament_export_targets(export_navigation_core HAS_LIBRARY_TARGET)
ament_export_dependencies(eigen3_cmake_module Eigen3 terramind_transport)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(navigation_test test/navigation_test.cpp)
  target_link_libraries(navigation_test navigation_core)
  ament_add_gtest(navigation_discovery_test test/discovery_test.cpp)
  target_link_libraries(navigation_discovery_test navigation_discovery util Threads::Threads)
endif()
ament_package()
