cmake_minimum_required(VERSION 3.16)
# 同时安装控制管理节点并导出 CommandGuard，供两种后端复用保护逻辑。
project(terramind_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(terramind_protocol REQUIRED)
find_package(terramind_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)

add_library(${PROJECT_NAME} SHARED src/command_guard.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} terramind_protocol terramind_interfaces rclcpp std_srvs geometry_msgs)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(terramind_protocol terramind_interfaces rclcpp std_srvs geometry_msgs)
add_executable(control_manager_node src/control_manager_node.cpp)
target_link_libraries(control_manager_node ${PROJECT_NAME})
ament_target_dependencies(control_manager_node rclcpp std_srvs geometry_msgs terramind_interfaces)
install(TARGETS control_manager_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(guard_test test/guard_test.cpp)
target_link_libraries(guard_test ${PROJECT_NAME})
endif()

ament_package()
