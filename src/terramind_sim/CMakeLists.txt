cmake_minimum_required(VERSION 3.16)
# 共享行为/运动模型，分别供 ROS 直接仿真与普通 PTY 模拟器进程调用。
project(terramind_sim)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(terramind_protocol REQUIRED)
find_package(terramind_transport REQUIRED)
find_package(terramind_control REQUIRED)
find_package(terramind_mcu REQUIRED)
find_package(terramind_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)

add_library(${PROJECT_NAME} SHARED src/board_model.cpp src/differential_drive_model.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} terramind_protocol terramind_transport terramind_control terramind_mcu terramind_interfaces rclcpp nav_msgs sensor_msgs tf2_ros diagnostic_msgs std_srvs)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(terramind_protocol terramind_transport terramind_control terramind_mcu terramind_interfaces rclcpp nav_msgs sensor_msgs tf2_ros diagnostic_msgs std_srvs)
add_executable(sim_interface_node src/sim_interface_node.cpp)
target_link_libraries(sim_interface_node ${PROJECT_NAME})
ament_target_dependencies(sim_interface_node rclcpp nav_msgs sensor_msgs tf2_ros diagnostic_msgs std_srvs)
add_executable(serial_board_emulator src/serial_board_emulator.cpp)
target_link_libraries(serial_board_emulator ${PROJECT_NAME} util)
install(TARGETS sim_interface_node serial_board_emulator DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS tools/serial_loopback_test.py tools/control_cli.py tools/ros_smoke_test.py tools/serial_discovery_test.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(model_test test/model_test.cpp)
target_link_libraries(model_test ${PROJECT_NAME})
endif()

ament_package()
